local function getRarityName(category, assets)
    local a = assets or getAssets()
    if not a or not a.Directory[category] then return "Unknown", 0 end
    local r = a.Directory[category].Rarity
    if type(r) ~= "table" then return "Unknown", 0 end
    return r.DisplayName or r._id or r.Name or "Unknown", tonumber(r.RarityNumber) or 0
end

local function getEggsList()
    local a = getAssets()
    if not a then return {} end
    local eggsFolder = game:GetService("ReplicatedStorage"):FindFirstChild("Assets")
    local models = eggsFolder and eggsFolder:FindFirstChild("Models")
    local eggModels = models and models:FindFirstChild("Eggs")
    local list = {}
    for name, data in pairs(a.Directory) do
        if type(name) == "string" and type(data) == "table" and type(data.Rarity) == "table" then
            local valid = true
            if eggModels then valid = eggModels:FindFirstChild(name) ~= nil end
            if valid then
                table.insert(list, { Name = name, Rarity = data.Rarity.DisplayName or data.Rarity._id or data.Rarity.Name or "Unknown", RarityNum = tonumber(data.Rarity.RarityNumber) or 0 })
            end
        end
    end
    table.sort(list, function(x, y)
        if x.RarityNum ~= y.RarityNum then return x.RarityNum > y.RarityNum end
        return x.Name < y.Name
    end)
    local names = {}
    local lookup = {}
    for _, v in ipairs(list) do
        local display = "[" .. v.Rarity .. "] " .. v.Name
        table.insert(names, display)
        lookup[display] = v.Name
    end
    BluzHub.Farm.EggLookup = lookup
    return names
end

local function getRaritiesList()
    local a = getAssets()
    if not a then return {} end
    local seen = {}
    for _, data in pairs(a.Directory) do
        if type(data) == "table" and type(data.Rarity) == "table" then
            local num = tonumber(data.Rarity.RarityNumber) or 0
            local name = data.Rarity.DisplayName or data.Rarity._id or data.Rarity.Name
            if type(name) == "string" and name ~= "" then
                if not seen[name] or num > seen[name] then seen[name] = num end
            end
        end
    end
    local arr = {}
    for name, num in pairs(seen) do table.insert(arr, { Name = name, Num = num }) end
    table.sort(arr, function(x, y)
        if x.Num ~= y.Num then return x.Num > y.Num end
        return x.Name < y.Name
    end)
    local out = {}
    for _, v in ipairs(arr) do table.insert(out, v.Name) end
    return out
end

local function getAreasList()
    local dir = game:GetService("ReplicatedStorage"):FindFirstChild("Directory")
    local areas = dir and dir:FindFirstChild("Areas")
    local a = getAssets()
    local list = {}
    local seen = {}
    if areas and areas:IsA("ModuleScript") then
        local ok, mod = pcall(require, areas)
        if ok and type(mod) == "table" and type(mod.Directory) == "table" then
            for id, data in pairs(mod.Directory) do
                if type(id) == "string" and type(data) == "table" then
                    local maxRarity = 0
                    if type(data.DropTable) == "table" then
                        for _, drop in ipairs(data.DropTable) do
                            if type(drop) == "table" and drop[1] and a and a.Directory[drop[1]] then
                                local r = a.Directory[drop[1]].Rarity
                                if type(r) == "table" then
                                    local num = tonumber(r.RarityNumber) or 0
                                    if num > maxRarity then maxRarity = num end
                                end
                            end
                        end
                    end
                    table.insert(list, { Id = id, Name = data.DisplayName or id, RarityNum = maxRarity })
                    seen[id] = true
                end
            end
        end
    end
    local cmds = getEggCmds()
    if cmds then
        local ok, snap = pcall(function() return cmds.GetAreaEggSnapshot() end)
        if ok and type(snap) == "table" and type(snap.Records) == "table" then
            for _, rec in ipairs(snap.Records) do
                if type(rec.AreaId) == "string" and not seen[rec.AreaId] then
                    seen[rec.AreaId] = true
                    table.insert(list, { Id = rec.AreaId, Name = rec.AreaId, RarityNum = 0 })
                end
            end
        end
    end
    table.sort(list, function(x, y)
        if x.RarityNum ~= y.RarityNum then return x.RarityNum > y.RarityNum end
        return x.Name < y.Name
    end)
    local names = {}
    local lookup = {}
    for _, v in ipairs(list) do
        lookup[v.Id] = v.Name
        table.insert(names, v.Name)
    end
    BluzHub.Farm.AreasNameById = lookup
    return names
end

end

local function findBestTarget()
    BluzHub.Farm.FailCounts = BluzHub.Farm.FailCounts or {}
    local cmds = getEggCmds()
    if not cmds then return nil end
    local ok, snap = pcall(function() return cmds.GetAreaEggSnapshot() end)
    if not ok or type(snap) ~= "table" or type(snap.Records) ~= "table" then return nil end
    local assets = getAssets()
    local candidates = {}
    for _, rec in ipairs(snap.Records) do
        if rec.Uid and rec.AssetCategory and rec.CarrierUserId == nil then
            if rec.State == "Slot" or rec.State == "Dropped" then
                local _, rarityNum = getRarityName(rec.AssetCategory, assets)
                table.insert(candidates, { Record = rec, RarityNum = rarityNum, Uid = rec.Uid })
            end
        end
    end
    if #candidates == 0 then return nil end
    local filtered = {}
    for _, v in ipairs(candidates) do
        if (BluzHub.Farm.FailCounts[v.Uid] or 0) < 2 then table.insert(filtered, v) end
    end
    if #filtered == 0 then
        BluzHub.Farm.FailCounts = {}
        filtered = candidates
    end
    local function bestOf(arr)
        if #arr == 0 then return nil end
        local best = arr[1]
        for i = 2, #arr do
            local v = arr[i]
            if v.RarityNum > best.RarityNum or (v.RarityNum == best.RarityNum and tostring(v.Uid) < tostring(best.Uid)) then best = v end
        end
        return best
    end
    if #BluzHub.Farm.SpecificEggNames > 0 then
        local set = {}
        for _, n in ipairs(BluzHub.Farm.SpecificEggNames) do if type(n) == "string" then set[n] = true end end
        local match = {}
        for _, v in ipairs(filtered) do if set[v.Record.AssetCategory] then table.insert(match, v) end end
        if #match > 0 then return bestOf(match) end
    end
    if #BluzHub.Farm.SelectedRarities > 0 then
        local set = {}
        for _, r in ipairs(BluzHub.Farm.SelectedRarities) do if type(r) == "string" then set[r] = true end end
        local match = {}
        for _, v in ipairs(filtered) do if set[getRarityName(v.Record.AssetCategory, assets)] then table.insert(match, v) end end
        if #match > 0 then return bestOf(match) end
    end
    if #BluzHub.Farm.SelectedAreas > 0 then
        local set = {}
        for _, a in ipairs(BluzHub.Farm.SelectedAreas) do if type(a) == "string" then set[a] = true end end
        local match = {}
        for _, v in ipairs(filtered) do
            local areaName = BluzHub.Farm.AreasNameById[v.Record.AreaId] or v.Record.AreaId
            if areaName and set[areaName] then table.insert(match, v) end
        end
        if #match > 0 then return bestOf(match) end
    end
    return nil
end

local function stealEgg(record)
    if not record or not record.Uid then return false end
    BluzHub.Farm.FailCounts = BluzHub.Farm.FailCounts or {}
    local cmds = getEggCmds()
    if not cmds then return false end
    local ok, snap = pcall(function() return cmds.GetAreaEggSnapshot() end)
    if not ok or type(snap) ~= "table" then return false end
    local pos
    local found = false
    for _, rec in ipairs(snap.Records) do
        if rec.Uid == record.Uid then
            if rec.CarrierUserId == nil and (rec.State == "Slot" or rec.State == "Dropped") then
                found = true
                if rec.BottomCFrame then
                    local ok2, p = pcall(function() return rec.BottomCFrame.Position end)
                    if ok2 and p then pos = p end
                end
            end
            break
        end
    end
    if not found then BluzHub.Farm.FailCounts[record.Uid] = 999; return false end
    if not pos then
        local model = findEggModel(record.Uid)
        if model then
            local ok2, p = pcall(function() return model:GetPivot().Position end)
            if ok2 and p then pos = p end
        end
    end
    if not pos then return false end
    BluzHub.Farm.LastTargetUid = record.Uid
    teleport(pos + Vector3.new(0, 2, 0))
    pcall(function() cmds.RequestCarryAreaEgg(record.Uid) end)
    local model = findEggModel(record.Uid)
    if model and fireproximityprompt then
        local prompt = model:FindFirstChild("CarryAreaEgg", true)
        if prompt and prompt:IsA("ProximityPrompt") then pcall(fireproximityprompt, prompt) end
    end
    local timeout = tick() + 1
    while tick() < timeout do
        if isCarrying() then BluzHub.Farm.FailCounts[record.Uid] = nil; return true end
        task.wait(0.03)
    end
    teleport(pos + Vector3.new(0, 2, 0))
    pcall(function() cmds.RequestCarryAreaEgg(record.Uid) end)
    if model and fireproximityprompt then
        local prompt = model:FindFirstChild("CarryAreaEgg", true)
        if prompt and prompt:IsA("ProximityPrompt") then pcall(fireproximityprompt, prompt) end
    end
    local timeout2 = tick() + 0.5
    while tick() < timeout2 do
        if isCarrying() then BluzHub.Farm.FailCounts[record.Uid] = nil; return true end
        task.wait(0.03)
    end
    if not isCarrying() then BluzHub.Farm.FailCounts[record.Uid] = (BluzHub.Farm.FailCounts[record.Uid] or 0) + 1 end
    return isCarrying()
end

local function returnToBase()
    local plotCmds = getPlotCmds()
    if not plotCmds then return false end
    local ok, data = pcall(function() return plotCmds.GetPlotData() end)
    if not ok or type(data) ~= "table" then return false end
    local pos
    if data.PetArea and data.PetArea.Position then pos = data.PetArea.Position
    elseif data.CenterPoint and data.CenterPoint.CFrame then pos = data.CenterPoint.CFrame.Position
    elseif data.CenterPoint and data.CenterPoint.Position then pos = data.CenterPoint.Position end
    if not pos then return false end
    teleport(pos + Vector3.new(0, 5, 0))
    task.wait(0.03)
    if isCarrying() then teleport(pos + Vector3.new(0, 5, 0)) end
    return true
end

local function waitClaim()
    local timeout = tick() + 2.5
    while tick() < timeout do
        if not isCarrying() then return true end
        task.wait(0.03)
    end
    return not isCarrying()
end

local function farmRunOnce()
    if not BluzHub.Farm.Enabled then return false, "disabled" end
    if isCarrying() then returnToBase(); waitClaim(); return true, "claimed" end
    local target = findBestTarget()
    if not target then return false, "no target" end
    if stealEgg(target.Record) then returnToBase(); waitClaim(); return true, "stolen" end
    return false, "steal failed"
end

local function startAutoFarm()
    if BluzHub.Farm.Running then return end
    BluzHub.Farm.FailCounts = {}
    BluzHub.Farm.Enabled = true
    BluzHub.Farm.Running = true
    BluzHub.Farm.Loop = task.spawn(function()
        while BluzHub.Farm.Enabled do
            local ok, res = pcall(farmRunOnce)
            if not ok then task.wait(0.2) elseif res == false then task.wait(0.1) end
            task.wait(0.03)
        end
        BluzHub.Farm.Running = false
    end)
    addTask(BluzHub.Farm.Loop)
end

local function stopAutoFarm()
    BluzHub.Farm.Enabled = false
    BluzHub.Farm.Running = false
    BluzHub.Farm.FailCounts = {}
    BluzHub.Farm.LastTargetUid = nil
    if BluzHub.Farm.Loop then task.cancel(BluzHub.Farm.Loop); BluzHub.Farm.Loop = nil end
end

            local ok, mod = pcall(require, save)

        end
    end

            if ok and type(mod) == "table" then deps.Network = mod end
        end
    end
    if not deps.EggItemUtil and util then
        local eggItem = util:FindFirstChild("EggItemUtil")
        if eggItem and eggItem:IsA("ModuleScript") then
            local ok, mod = pcall(require, eggItem)
            if ok and type(mod) == "table" then deps.EggItemUtil = mod end

    end
    if not deps.AssetItemSerialization and util then

            local ok, mod = pcall(require, serial)
            if ok and type(mod) == "table" then deps.AssetItemSerialization = mod end

    end
    if not deps.GetPriceValueForItem and util then
        local price = util:FindFirstChild("GetPriceValueForItem")
        if price and price:IsA("ModuleScript") then
            local ok, mod = pcall(require, price)
            if ok and type(mod) == "function" then deps.GetPriceValueForItem = mod end

local function getSellableEggs()
    local deps = sellLoadDeps()
    if not deps.Save or not deps.EggItemUtil then return {} end
    local ok, save = pcall(function() return deps.Save.Get() end)
    if not ok or type(save) ~= "table" then return {} end
    local inventory = save.EggInventory or {}
    local assets = getAssets()
    local list = {}
    for uid, item in pairs(inventory) do
        if type(item) == "table" and item.Placement == nil then
            local ok2, egg = pcall(function() return deps.EggItemUtil.DeserializeSavedEgg(item) end)
            if ok2 and type(egg) == "table" then
                local category = egg.AssetCategory
                local ok3, price = pcall(function() return deps.EggItemUtil.GetSellPrice(egg) end)
                local rarityName, rarityNum = "Unknown", 0
                if assets and assets.Directory[category] then
                    local r = assets.Directory[category].Rarity
                    if type(r) == "table" then rarityName = r.DisplayName or r._id or r.Name or "Unknown"; rarityNum = tonumber(r.RarityNumber) or 0 end
                end
                table.insert(list, { uid = uid, category = category, price = ok3 and math.floor(price) or 0, rarity = rarityName, rarityNum = rarityNum })
            end
        end
    end
    return list
end

local function getSellablePets()
    local deps = sellLoadDeps()
    if not deps.Save or not deps.AssetItemSerialization or not deps.GetPriceValueForItem then return {} end
    local ok, save = pcall(function() return deps.Save.Get() end)
    if not ok or type(save) ~= "table" then return {} end
    local inventory = save.Inventory or {}
    local equipped = save.EquippedAssets or {}
    local favorites = save.FavoriteAssets or {}
    local assets = getAssets()
    local vipMult = LocalPlayer:GetAttribute("VIP") and 2 or 1
    local list = {}
    for uid, item in pairs(inventory) do
        if type(item) == "table" then
            if not table.find(equipped, uid) and not favorites[uid] then
                local ok2, pet = pcall(function() return deps.AssetItemSerialization.Deserialize(item) end)
                if ok2 and type(pet) == "table" then
                    if not pet.IsFavorite and not pet.InFuse then
                        local category = pet.Category
                        local ok3, price = pcall(function() return deps.GetPriceValueForItem(item) * 1.33 * vipMult end)
                        local rarityName, rarityNum = "Unknown", 0
                        if assets and assets.Directory[category] then
                            local r = assets.Directory[category].Rarity
                            if type(r) == "table" then rarityName = r.DisplayName or r._id or r.Name or "Unknown"; rarityNum = tonumber(r.RarityNumber) or 0 end
                        end
                        table.insert(list, { uid = uid, category = category, price = ok3 and math.floor(price) or 0, rarity = rarityName, rarityNum = rarityNum })
                    end
                end
            end
        end
    end
    return list
end

            end
        end
    end
    table.sort(list, function(x, y)
        if x.rarityNum ~= y.rarityNum then return x.rarityNum > y.rarityNum end
        return x.name < y.name
    end)
    local names = {}
    local lookup = {}
    for _, v in ipairs(list) do
        local display = "[" .. v.rarity .. "] " .. v.name
        table.insert(names, display)
        lookup[display] = v.name
    end
    return names, lookup
end

local function filterByCategories(items, selected)
    if not items or #items == 0 or not selected or #selected == 0 then return {} end
    local set = {}
    for _, v in ipairs(selected) do
        if type(v) == "string" then
            local clean = v:match("^%[[^%]]+%]%s*(.+)$")
            if clean and clean ~= "" then set[clean] = true end
        end
    end
    local out = {}
    for _, item in ipairs(items) do if set[item.category] then table.insert(out, item) end end
    return out
end

local function buyBestTrail()
    local trails = (function()
        local dir = game:GetService("ReplicatedStorage"):FindFirstChild("Directory")
        local t = dir and dir:FindFirstChild("Trails")
        if not t or not t:IsA("ModuleScript") then return nil end
        local ok, mod = pcall(require, t)
        if ok and type(mod) == "table" and type(mod.Directory) == "table" then return mod end
        return nil
    end)()
    local save = getSaveModule()
    local network = getNetworkModule()
    if not trails or not save or not network then return false, "missing" end
    if not network.NET_MAP or not network.NET_MAP.Trails then return false, "no netmap" end
    local ok, data = pcall(function() return save.Get() end)
    if not ok or type(data) ~= "table" then return false, "no save" end
    local money = tonumber(data.Money) or 0
    local owned = data.TrailInventory or {}
    local equipped = data.EquippedTrail
    local bestOwnedSpeed = 0
    local bestOwnedId
    for id, val in pairs(owned) do
        if val and trails.Directory[id] then
            local speed = tonumber(trails.Directory[id].SpeedMultiplier) or 1
            if speed > bestOwnedSpeed then bestOwnedSpeed = speed; bestOwnedId = id end
        end
    end
    local bestBuy
    local bestBuySpeed = bestOwnedSpeed
    for id, info in pairs(trails.Directory) do
        if type(id) == "string" and type(info) == "table" then
            if info.DisplayInShop and not info.RobuxOnly then
                if not owned[id] then
                    local price = tonumber(info.Price) or 0
                    local speed = tonumber(info.SpeedMultiplier) or 1
                    if price > 0 and price <= money and speed > bestBuySpeed then
                        bestBuySpeed = speed
                        bestBuy = { Id = id, Data = info, Speed = speed }
                    end
                end
            end
        end
    end
    local trailsNet = network.NET_MAP.Trails
    if bestBuy then
        pcall(function() network.Invoke(trailsNet.REQUEST_PURCHASE, bestBuy.Id) end)
        task.wait(0.35)
        pcall(function() network.Invoke(trailsNet.REQUEST_SELECT, bestBuy.Id) end)
        return true, "bought " .. bestBuy.Id
    end
    if bestOwnedId and equipped ~= bestOwnedId then
        pcall(function() network.Invoke(trailsNet.REQUEST_SELECT, bestOwnedId) end)
        return true, "equipped " .. bestOwnedId
    end
    return false, "up to date"
end

local function buyBestBike()
    local bikeCatalog = (function()
        local system = game:GetService("ReplicatedStorage"):FindFirstChild("OurBikeSystem")
        if not system then return nil end
        local catalog = system:FindFirstChild("Catalog") or system:WaitForChild("Catalog", 5)
        if not catalog or not catalog:IsA("ModuleScript") then return nil end
        local ok, mod = pcall(require, catalog)
        if ok and type(mod) == "table" and #mod > 0 then return mod end
        return nil
    end)()
    local shopRequest = (function()
        local system = game:GetService("ReplicatedStorage"):FindFirstChild("OurBikeSystem")
        if not system then return nil end
        local req = system:FindFirstChild("ShopRequest") or system:WaitForChild("ShopRequest", 5)
        if req and req:IsA("RemoteFunction") then return req end
        return nil
    end)()
    local save = getSaveModule()
    if not bikeCatalog or not shopRequest then return false, "missing" end
    local money = 0
    if save then
        local ok, data = pcall(function() return save.Get() end)
        if ok and type(data) == "table" then money = tonumber(data.Money) or 0 end
    end
    local ok, state = pcall(function() return shopRequest:InvokeServer("State") end)
    if not ok or type(state) ~= "table" then return false, "state failed" end
    local owned = state.Owned or {}
    local equipped = state.Equipped or "Bicycle"
    if type(state.Money) == "number" and state.Money > money then money = state.Money end
    local cashBikes = {}
    for _, bike in ipairs(bikeCatalog) do
        if type(bike) == "table" and type(bike.Id) == "string" then
            if not bike.AdminOnly and not bike.RobuxOnly then table.insert(cashBikes, bike) end
        end
    end
    if #cashBikes == 0 then return false, "no cash bikes" end
    table.sort(cashBikes, function(a, b) return (tonumber(a.SpeedMultiplier) or 1) > (tonumber(b.SpeedMultiplier) or 1) end)
    local bestOwnedSpeed = 1
    for _, bike in ipairs(cashBikes) do
        if owned[bike.Id] == true then
            local speed = tonumber(bike.SpeedMultiplier) or 1
            if speed > bestOwnedSpeed then bestOwnedSpeed = speed end
        end
    end
    for _, bike in ipairs(cashBikes) do
        local speed = tonumber(bike.SpeedMultiplier) or 1
        local price = tonumber(bike.Price) or 0
        if owned[bike.Id] ~= true and speed > bestOwnedSpeed and price > 0 and price <= money then
            local ok2, res = pcall(function() return shopRequest:InvokeServer("Buy", bike.Id) end)
            if ok2 and type(res) == "table" and res.Ok then
                notify("Bike Shop", "Bought: " .. bike.Id, 3)
                return true, "bought " .. bike.Id
            end
            task.wait(0.2)
        end
    end
    local bestOwnedId
    local bestSpeed = 1
    for _, bike in ipairs(cashBikes) do
        if owned[bike.Id] == true then
            local speed = tonumber(bike.SpeedMultiplier) or 1
            if speed > bestSpeed then bestSpeed = speed; bestOwnedId = bike.Id end
        end
    end
    if bestOwnedId and equipped ~= bestOwnedId then
        if pcall(function() return shopRequest:InvokeServer("Equip", bestOwnedId) end) then
            return true, "equipped " .. bestOwnedId
        end
    end
    return false, "up to date"
end

local function plotParseSelection(selection)
    local out = {}
    for _, v in ipairs(selection or {}) do
        if type(v) == "string" then
            if v == "All" or v == "Best" then table.insert(out, v)
            else
                local clean = v:match("^%[[^%]]+%]%s*(.+)$")
                if clean and clean ~= "" then table.insert(out, clean) end
            end
        end
    end
    return out
end

local function plotMatchesSelection(category, selection)
    local parsed = plotParseSelection(selection)
    if #parsed == 0 then return false end
    for _, v in ipairs(parsed) do
        if v == "All" or v == "Best" then return true end
        if v == category then return true end
    end
    return false
end

local function plotGetFreePlacement(plotData)
    if not plotData or not plotData.PetArea or not plotData.CenterPoint then return nil end
    local petArea = plotData.PetArea
    if not petArea:IsA("BasePart") then return nil end
    local cmds = getEggCmds()
    if not cmds then return nil end
    local ok, records = pcall(function() return cmds.GetOwnerRuntimeRecords(LocalPlayer.UserId) end)
    local placed = {}
    if ok and type(records) == "table" then
        for _, rec in pairs(records) do
            if rec.Placement and rec.Placement.LocalCFrame then
                local cf = rec.Placement.LocalCFrame
                if typeof(cf) == "CFrame" then
                    local pos = (plotData.CenterPoint.CFrame * cf).Position
                    table.insert(placed, Vector3.new(pos.X, 0, pos.Z))
                end
            end
        end
    end
    local size = petArea.Size
    local cf = petArea.CFrame
    local cols = math.max(math.floor((size.X - 4) / 4), 1)
    local rows = math.max(math.floor((size.Z - 4) / 4), 1)
    for row = 0, rows - 1 do
        for col = 0, cols - 1 do
            local pos = (cf * CFrame.new(-((cols - 1) * 4) / 2 + col * 4, size.Y / 2, -((rows - 1) * 4) / 2 + row * 4)).Position
            local occupied = false
            for _, p in ipairs(placed) do
                local dx = pos.X - p.X
                local dz = pos.Z - p.Z
                if math.sqrt(dx * dx + dz * dz) < 3.4 then occupied = true; break end
            end
            if not occupied then return plotData.CenterPoint.CFrame:ToObjectSpace(CFrame.new(pos)) end
        end
    end
    return nil
end

local function plotPlaceEggTool(tool, plotData)
    if not tool or not tool.Parent then return false, "no tool" end
    if not plotData then return false, "no plot" end
    local uid = tool:GetAttribute("UID")
    if type(uid) ~= "string" or uid == "" then return false, "bad uid" end
    local placeRemote = getNetworkRemote("Eggs: RequestPlaceEgg")
    local equipRemote = getNetworkRemote("Eggs: RequestEquipTool")
    if not placeRemote then return false, "no place remote" end
    if not equipRemote then return false, "no equip remote" end
    local localCf = plotGetFreePlacement(plotData)
    if not localCf then return false, "no space" end
    local char = LocalPlayer.Character
    if not char then return false, "no character" end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return false, "no humanoid" end
    pcall(function() humanoid:UnequipTools() end)
    task.wait(0.1)
    local ok, result = pcall(function() return equipRemote:InvokeServer(uid) end)
    if not ok then return false, "equip invoke error" end
    if result ~= true then return false, "equip rejected: " .. tostring(result) end
    local foundTool
    local timeout = tick() + 2.5
    while tick() < timeout do
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") and child:GetAttribute("ItemType") == "AssetEgg" and child:GetAttribute("UID") == uid then foundTool = child; break end
        end
        if foundTool then break end
        task.wait(0.05)
    end
    if not foundTool then return false, "tool did not appear" end
    task.wait(0.2)
    local ok2, result2 = pcall(function() return placeRemote:InvokeServer({ Uid = uid, LocalCFrame = localCf }) end)
    task.wait(0.15)
    if foundTool.Parent == char then pcall(function() humanoid:UnequipTools() end) end
    if not ok2 then return false, "place invoke error" end
    if result2 ~= true then return false, "place rejected: " .. tostring(result2) end
    return true, "placed"
end

    if not ok2 or type(save) ~= "table" then return false, "no save" end
    local inventory = save.EggInventory or {}
    local assets = getAssets()
    local allowAll = false
    local categorySet = {}
    for _, v in ipairs(plotParseSelection(BluzHub.Plot.PlaceSelectedEggs)) do
        if v == "Best" or v == "All" then allowAll = true else categorySet[v] = true end
    end
    local candidates = {}
    for uid, item in pairs(inventory) do
        if type(item) == "table" and item.Placement == nil then
            local ok3, egg = pcall(function() return deps.EggItemUtil.DeserializeSavedEgg(item) end)
            if ok3 and type(egg) == "table" then
                local category = egg.AssetCategory
                if category and (allowAll or categorySet[category]) then
                    local rarityNum, luck = 0, 0
                    if assets and assets.Directory[category] then
                        local r = assets.Directory[category].Rarity
                        if type(r) == "table" then rarityNum = tonumber(r.RarityNumber) or 0 end
                        luck = tonumber(assets.Directory[category].Luck) or 0
                    end
                    table.insert(candidates, { uid = uid, cat = category, score = rarityNum, luck = luck })
                end
            end
        end
    end
    if #candidates == 0 then return false, "no matching unplaced eggs" end
    table.sort(candidates, function(a, b)
        if a.score ~= b.score then return a.score > b.score end
        if a.luck ~= b.luck then return a.luck > b.luck end
        return a.cat < b.cat
    end)
    local best = candidates[1]
    local localCf = plotGetFreePlacement(plotData)
    if not localCf then return false, "no free spot" end
    local equipRemote = getNetworkRemote("Eggs: RequestEquipTool")
    local placeRemote = getNetworkRemote("Eggs: RequestPlaceEgg")
    if not equipRemote or not placeRemote then return false, "no remotes" end
    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false, "no humanoid" end
    pcall(function() humanoid:UnequipTools() end)
    task.wait(0.15)
    local ok4, result = pcall(function() return equipRemote:InvokeServer(best.uid) end)
    if not ok4 then return false, "equip error" end
    if not result then return false, "equip rejected" end
    local found = false
    local timeout = tick() + 4
    while tick() < timeout do
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") and child:GetAttribute("ItemType") == "AssetEgg" and child:GetAttribute("UID") == best.uid then found = true; break end
        end
        if found then break end
        task.wait(0.05)
    end
    if not found then pcall(function() humanoid:UnequipTools() end); return false, "tool not in character" end
    task.wait(0.2)
    local ok5, result2 = pcall(function() return placeRemote:InvokeServer({ Uid = best.uid, LocalCFrame = localCf }) end)
    task.wait(0.15)
    pcall(function() humanoid:UnequipTools() end)
    if not ok5 then return false, "place error" end
    if not result2 then return false, "place rejected" end
    return true, "placed " .. best.cat
end

local function plotGetTreadmillPart()
    local plotCmds = getPlotCmds()
    if not plotCmds then return nil end
    local ok, plotData = pcall(function() return plotCmds.GetPlotData() end)
    if not ok or type(plotData) ~= "table" or not plotData.PlotFolder then return nil end
    local part = plotData.PlotFolder:FindFirstChild("TreadmillBottom")
    if part and part:IsA("BasePart") then return part end
    local part2 = plotData.PlotFolder:FindFirstChild("TreadmillBottom", true)
    if part2 and part2:IsA("BasePart") then return part2 end
    return nil
end

local function plotEquipTreadmill()
    local part = plotGetTreadmillPart()
    if not part then return false, "no TreadmillBottom" end
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false, "no character" end
    local target = part.Position + Vector3.new(0, part.Size.Y / 2 + 3, 0)
    local look = hrp.CFrame.LookVector
    pcall(function()
        hrp.CFrame = CFrame.new(target, target + look)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    task.wait(0.3)
    local network = getNetworkModule()
    if network and network.NET_MAP and network.NET_MAP.Treadmills then
        pcall(function() network.Invoke(network.NET_MAP.Treadmills.REQUEST_EQUIP_STATIC) end)
    end
    return true
end

local function plotExitTreadmill()
    local network = getNetworkModule()
    if network and network.NET_MAP and network.NET_MAP.Treadmills then
        pcall(function() network.Invoke(network.NET_MAP.Treadmills.REQUEST_UNEQUIP) end)
    end
    task.wait(0.2)
    local plotCmds = getPlotCmds()
    if plotCmds then
        local ok, data = pcall(function() return plotCmds.GetPlotData() end)
        if ok and type(data) == "table" then
            local area = data.PetArea or data.CenterPoint
            if area and area.Position then
                local char = LocalPlayer.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local target = area.Position + Vector3.new(0, 5, 0)
                    local look = hrp.CFrame.LookVector
                    pcall(function()
                        hrp.CFrame = CFrame.new(target, target + look)
                        hrp.AssemblyLinearVelocity = Vector3.zero
                        hrp.AssemblyAngularVelocity = Vector3.zero
                    end)
                end
            end
        end
    end
    return true
end

local function plotClaimIndexRewards()
    local network = getNetworkModule()
    if not network or not network.NET_MAP or not network.NET_MAP.Index then return false end
    local index = network.NET_MAP.Index
    if not index.REQUEST_CLAIM_ALL then return false end
    local ok, result = pcall(function() return network.Invoke(index.REQUEST_CLAIM_ALL) end)
    return ok and result
end
